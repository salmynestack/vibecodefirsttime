id integer auto increment

created_at timestamp default current_timestamp

request body :
{
    "name": "denis",
    "email": "denis@localhost",
    "password": "jagoo"
}

response body  (success) :

{
    "data" : "token"
}

response body (error) :

{
    "error" : "email atau password salah"
}
